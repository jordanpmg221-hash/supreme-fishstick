    function updateClock() {
      const now = new Date();
      document.getElementById('clock').innerText = now.toLocaleTimeString('fr-FR');

      // Program range: Sept 1, 2026 to Mar 31, 2027
      const startDate = new Date('2026-09-01T00:00:00');
      const endDate = new Date('2027-03-31T23:59:59');
      
      let pct = 0;
      if (now >= startDate && now <= endDate) {
        pct = Math.round(((now - startDate) / (endDate - startDate)) * 100);
      } else if (now > endDate) {
        pct = 100;
      }
      document.getElementById('program-bar').style.width = pct + '%';
      document.getElementById('program-percent').innerText = pct + '%';
    }

                else if (item.risqLevel === 'Élevé') badgeClass = "bg-amber-500/20 text-amber-400 border-amber-500/30";
                else if (item.risqLevel === 'Moyen') badgeClass = "bg-yellow-500/10 text-yellow-400 border-yellow-500/20";

                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-800/40 transition-colors";
                tr.innerHTML = `
                    <td class="p-4 font-mono text-slate-400 font-medium">${item.id}</td>
                    <td class="p-4 text-slate-300">${item.date}</td>
                    <td class="p-4 font-semibold text-slate-200">${item.libelle}</td>
                    <td class="p-4 font-mono font-medium text-slate-100">${item.montant.toLocaleString()} FCFA</td>
                    <td class="p-4">
                        <span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-[10px] font-semibold border ${badgeClass}">
                            ${item.risqLevel}
                        </span>
                    </td>
                    <td class="p-4 text-slate-400 max-w-xs leading-relaxed">${item.motif}</td>
                    <td class="p-4 text-right">
                        <button onclick="resolveAudit('${item.id}')" class="px-2.5 py-1 bg-slate-800 hover:bg-slate-700 text-slate-300 border border-slate-700 rounded-lg text-xs font-medium transition-all">
                            Réviser
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function filterAudit(level) {
            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.classList.remove('bg-emerald-500/20', 'text-emerald-400', 'border-emerald-500/30');
                btn.classList.add('bg-slate-800', 'text-slate-400', 'border-slate-700');
            });

            if (event && event.currentTarget) {
                event.currentTarget.classList.remove('bg-slate-800', 'text-slate-400', 'border-slate-700');
                event.currentTarget.classList.add('bg-emerald-500/20', 'text-emerald-400', 'border-emerald-500/30');
            }

            if (level === 'all') {
                renderAuditTable(auditData);
            } else {
                const filtered = auditData.filter(item => item.risqLevel === level);
                renderAuditTable(filtered);
            }
        }

        function triggerOCR() {
            const ocrStatus = document.getElementById('ocrStatus');
            const container = document.getElementById('ocrResultContainer');
            const btn = document.getElementById('btnValidateEntry');

            if (ocrStatus) {
                ocrStatus.textContent = "Analyse IA en cours...";
                ocrStatus.className = "text-[10px] font-medium bg-amber-500/20 text-amber-400 px-2.5 py-1 rounded-full border border-amber-500/30 animate-pulse";
            }

            setTimeout(() => {
                if (ocrStatus) {
                    ocrStatus.textContent = "Extrait & Codifié (98.4%)";
                    ocrStatus.className = "text-[10px] font-medium bg-emerald-500/20 text-emerald-400 px-2.5 py-1 rounded-full border border-emerald-500/30";
                }
                if (container) {
                    container.classList.remove('opacity-50');
                }
                if (btn) {
                    btn.disabled = false;
                    btn.className = "w-full py-3 bg-emerald-500 hover:bg-emerald-600 text-slate-950 font-bold rounded-xl text-xs transition-all shadow-lg shadow-emerald-500/20 flex items-center justify-center space-x-2 cursor-pointer";
                }
            }, 1000);
        }
